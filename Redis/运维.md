```
import subprocess
import re
from collections import defaultdict
import time
import argparse
import redis

class RedisMonitor:
    def __init__(self, host='localhost', port=6379, password=None):
        self.host = host
        self.port = port
        self.password = password
        self.key_pattern = re.compile(r'^\d+\.\d+ \[(\d+) .*?\] ".*?" "(.*?)"')
        self.redis_client = redis.Redis(
            host=host,
            port=port,
            password=password,
            decode_responses=False
        )

    def start_monitor(self, output_file=None):
        """启动redis-cli monitor子进程并记录到文件"""
        if not output_file:
            output_file = f"redis_monitor_{int(time.time())}.log"

        cmd = ['redis-cli', '-h', self.host, '-p', str(self.port)]
        if self.password:
            cmd.extend(['-a', self.password])
        cmd.append('monitor')

        try:
            with open(output_file, 'w') as f:
                process = subprocess.Popen(
                    cmd,
                    stdout=subprocess.PIPE,
                    stderr=subprocess.PIPE,
                    encoding="utf-8"
                )

                print(f"Monitoring started. Logging to {output_file}")
                print("Press Ctrl+C to stop...")

                try:
                    for line in process.stdout:
                        f.write(line)
                        f.flush()
                except KeyboardInterrupt:
                    process.terminate()
                    print("\nMonitoring stopped.")

        except Exception as e:
            print(f"Error: {e}")

    def analyze_log(self, log_file):
        """分析日志文件并统计key请求"""
        stats = []

        try:
            with open(log_file, 'r') as f:
                for line in f:
                    match = self.key_pattern.search(line)
                    if match:
                        dbnum = match.group(1)
                        key = match.group(2)
                        stats.append((dbnum, key))

            # 统计每个key在db中的请求数
            counter = defaultdict(int)
            for dbnum, key in stats:
                counter[(dbnum, key)] += 1

            # 按请求数降序排序
            sorted_stats = sorted(counter.items(), key=lambda x: x[1], reverse=True)

            print("\nKey Request Statistics with Memory Usage (Top 30):")
            print("------------------------------------------------")
            print(f"{'Count':<8} {'Size':<10} {'DB':<4} {'Key':<40}")
            print("-" * 70)

            # 只处理前20个key
            for (dbnum, key), count in sorted_stats[:30]:
                try:
                    # 选择正确的数据库
                    self.redis_client.execute_command('SELECT', dbnum)
                    # 获取key大小(字节)
                    size = self.redis_client.memory_usage(key) or 0
                    size_str = f"{size/1024:.2f}KB" if size >= 1024 else f"{size}B"
                except Exception as e:
                    size_str = "N/A"

                print(f"{count:<8} {size_str:<10} {dbnum:<4} {key:<40}")

            return sorted_stats

        except Exception as e:
            print(f"Error analyzing log: {e}")
            return None

if __name__ == "__main__":
    parser = argparse.ArgumentParser(description='Redis Monitor Tool')
    parser.add_argument('--host', default='localhost', help='Redis host')
    parser.add_argument('--port', type=int, default=6379, help='Redis port')
    parser.add_argument('--password', help='Redis password')
    parser.add_argument('--log', help='Output log file path')
    parser.add_argument('--analyze', help='Analyze existing log file')

    args = parser.parse_args()

    monitor = RedisMonitor(host=args.host, port=args.port, password=args.password)

    if args.analyze:
        monitor.analyze_log(args.analyze)
    else:
        monitor.start_monitor(args.log)
```

4.keys新增内存解析

```
import os
from datetime import datetime, timedelta
import redis

# ==================================================
# 1. Redis 连接配置
# ==================================================
REDIS_HOST = "127.0.0.1"
REDIS_PORT = 6379
REDIS_USERNAME = None  # ACL 用户名；无则 None
REDIS_PASSWORD = os.getenv("REDIS_PASSWORD", "")
REDIS_DB = 0

# ==================================================
# 2. 筛选参数：通常只需要修改这里
# ==================================================

# 指定 Key 前缀，例如 "order:"、"user:"、"cache:"
KEY_PREFIX = "order:"

# 目标创建时间段的结束时间
TARGET_DATE = None          # None 表示今天；也可指定 "2026-10-16"
TARGET_END_TIME = "06:45:00"

# 向前统计多少小时
WINDOW_HOURS = 1

# Key 创建时设置的初始 TTL（小时）
INITIAL_TTL_HOURS = 72

# ==================================================
# 3. 扫描配置
# ==================================================
SCAN_COUNT = 500


def main():
    if not REDIS_PASSWORD:
        raise ValueError("请先设置 REDIS_PASSWORD 环境变量")

    # 计算目标时间段
    target_date = (
        datetime.strptime(TARGET_DATE, "%Y-%m-%d").date()
        if TARGET_DATE
        else datetime.now().date()
    )

    end_clock = datetime.strptime(
        TARGET_END_TIME, "%H:%M:%S"
    ).time()

    target_end = datetime.combine(target_date, end_clock)
    target_start = target_end - timedelta(hours=WINDOW_HOURS)
    now = datetime.now()

    # 根据目标时间段计算当前应剩余的 TTL
    initial_ttl = INITIAL_TTL_HOURS * 3600
    min_ttl = initial_ttl - (now - target_start).total_seconds()
    max_ttl = initial_ttl - (now - target_end).total_seconds()

    if max_ttl <= 0:
        print("目标 Key 的理论 TTL 已耗尽，无法通过当前 TTL 筛选。")
        return

    min_ttl = max(0, min_ttl)

    # Redis SCAN 的 MATCH 支持 glob 模式
    match_pattern = KEY_PREFIX + "*"

    print(f"Key 前缀：{KEY_PREFIX}")
    print(f"Key 匹配模式：{match_pattern}")
    print(f"目标创建时间段：{target_start} 至 {target_end}")
    print(f"当前时间：{now}")
    print(f"剩余 TTL 范围：{min_ttl:.0f}～{max_ttl:.0f} 秒")

    client = redis.Redis(
        host=REDIS_HOST,
        port=REDIS_PORT,
        username=REDIS_USERNAME,
        password=REDIS_PASSWORD,
        db=REDIS_DB,
        decode_responses=True,
        socket_connect_timeout=5,
        socket_timeout=10,
    )

    matched_count = 0
    total_bytes = 0
    missing_count = 0

    try:
        client.ping()
        print("\nRedis 连接成功，开始扫描...\n")

        # 只遍历指定前缀的 Key
        for key in client.scan_iter(
            match=match_pattern,
            count=SCAN_COUNT
        ):
            ttl = client.ttl(key)

            if min_ttl <= ttl <= max_ttl:
                size = client.memory_usage(key)

                if size is None:
                    missing_count += 1
                    continue

                matched_count += 1
                total_bytes += size

                print(
                    f"Key={key} | "
                    f"TTL={ttl}s | "
                    f"内存={size} bytes "
                    f"({size / 1024:.2f} KB)"
                )

        print("\n========== 统计结果 ==========")
        print(f"Key 前缀：{KEY_PREFIX}")
        print(f"符合条件的 Key 数量：{matched_count}")
        print(f"总内存：{total_bytes / 1024 / 1024:.2f} MB")
        print(f"无法获取内存的 Key 数量：{missing_count}")

        if matched_count:
            print(
                f"平均每个 Key："
                f"{total_bytes / matched_count / 1024:.2f} KB"
            )
        else:
            print("没有找到符合条件的 Key")

    finally:
        client.close()


if __name__ == "__main__":
    main()
```
